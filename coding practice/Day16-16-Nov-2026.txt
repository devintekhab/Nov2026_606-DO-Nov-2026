🔹 Jenkinsfile - deploy to MultiCluster envirtoment

pipeline {
    agent any

    stages {
        stage('Build') {
            steps {
                echo 'Building Docker image...'
                script {
                    docker.build('devintekhab-web-app')
                }
            }
        }

        stage('Test') {
            steps {
                echo 'Running tests...'
            }
        }

        stage('Deploy to Kubernetes') {
            steps {
                script {
                    if (env.CLUSTER == 'us-east') {
                        sh 'kubectl --context=us-east apply -f kubernets/deployment.yaml'
                    } else {
                        sh 'kubectl --context=us-west apply -f kubernets/deployment.yaml'
                    }
                }
            }
        }
    }
}
