Day 21 - Disaster recovery and backup stragetis

🔹 Ansible - backup and restore files

- name: backup and restore file
  hosts: all
  become: yes
  tasks:
   - name: Create a backup of /etc directory
     command: tar -czvf /tmp/etc-backup.tar.gz /etc
    
   - name: Copy backup to remote server
     copy:
       src: /tmp/etc-backup.tar.gz
       dest: /temp/etc-backup.tar.gz

   - name: restore backup from remote server
     command: tar -xzvf /tmp/etc-backup.tar.gz -C /

Jenkinsfile - backup stargety for jenkins workspace

pipeline {
    agent any {
       stages {
        stage("backup"){
            steps{
                echo 'backup jenkins workspace'
                sh 'tar -czvf /tmp/jenkins-backup.tar.gz- ${WORKSPACE}
            }
        }
        stage("Restore"){
            steps {
                echo 'Restore jenkins workspace'
                sh 'tar -xzvf /tmp/jenkins-backup.tar.gz -C ${WORKSPACE}
            }
        }
       }
    }
}
