🔹 Ansible - Deploy lambda functions using aws cli

name: Deploy AWS Lambda Functions

hosts: localhost

tasks:
  - name: Create Lambda Function
    command: >
      aws lambda create-function
      --function-name pathnex-lambda
      --runtime python3.8
      --role arn:aws:iam::12345678:role/lambda-execution-role-handler
      --handler lambda_functions.handler
      --zip-file fileb://lambda.zip

🔹 Jenkinsfile - Deploy Servless Application with lambda 

pipeline{
  agent any stages{
    stage("Build"){
       steps {
        echo "building lambda function ..."
        sh 'zip -r lambda.zip lambda/'
       }
    }
    stage('deploy to aws lambda'){
       steps {
        script {
          sh 'aws lambda update-function-code--function-name devintekhab-lambda --zip-file fileb://lambda.zip'
        }
       }
    }
  }
}
