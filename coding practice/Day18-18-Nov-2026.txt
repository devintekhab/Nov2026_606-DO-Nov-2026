🔹 Ansible - Configure HAProxy for load balancing

- name: Setup HAProxy Load Balancer
  hosts: all
  become: yes
  tsaks:
    - name: install HAProxy
      yum:
       name: haproxy
       state: present
    - name: configure HAProxy
      template: 
        src: haproxy.cfg.j2
        dest: /etc/haproxy/haproxy.cfg
    - name: Start HAPRoxy Service
      service:
        name: haproxy
        state: started
        enabled: yes

🔹 Kubernets - Horizontal pod autoscaler with multiple Replicas

apiVersion: app/v1
kind: deployment
metadata:
     name: devintekhab-web-app
spec: 
  replicas: 3
  selector: 
     matchlabels: 
        app: devintekhab-web-app
  template:
      metadata:
          labels:
            app: devintekhab-web-app
      spec:
        containers:
           -  name: ngnix
           -  image: ngnix
              ports:
                - containers: 80
  
---
apiVersion: autoscaling/v2
kind: HorizontalPodAutoscaler
metadata: 
   name: devintekhab-web-app-hpa
spec:
  scaleTragetRef:
     apiVersion: apps/v1

🔹 Jenkinsfile - Blue-green deployment stragety

pipeline {
    agent any {
        stages {
            stage("build"){
                steps {
                    echo "Building docker image"
                    docker.build("devintekhab-wb-app")
                }
            }
            stage("test"){
                steps {
                    echo "Runnnig test"
                }
            }
            stage("blue-green deployment"){
                steps {
                    script {
                        sh 'kubectl apply -f kubernets/blue-deployment.yaml
                        sh 'kubectl apply -f kubernets/green-deployment.yaml
                        sh 'kubectl rollout status deployment/blue-depoyment
                        sh 'kubectl rollout status deployment/green-deployment
                    }
                }
            }
        }
    }
}
