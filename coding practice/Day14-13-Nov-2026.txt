🔹 Ansible - setup ngnix with reverse proxy
- name: setup ngnix with reverse proxy
  hosts: all
  become: yes
  tasks:
    - name: install ngnix
      yum: 
        name: ngnix
        state: present
    - name: configure ngnix as reverse proxy
      template:
           src: reverse-proxy.conf.j2
           dest: /etc/ngnix/conf.d/reverse-proxy.conf
    - name: restart ngnix service
      service: 
         name: ngnix
         state: restatred
         enabled: yes

🔹 kubernetics - ingress with ngnix controller
apiVersion: networking.k8s.io/v1
kind: ingress
metadata:
    name: devintekhab-ingress
    annotations:
         ngnix.ingress.kubernetes.io/rewrite-target: /

spec: 
 rules:
 - hosts:
   http:
     paths: 
     - path: /
       pathType: Prefix
       backend: 
          service: 
            name: devintekhab-service
            port:
              number: 80

🔹  gitlab CI/CD - multi-stage deployment with helm

stages: 
  - build
  - push
  - deploy
  
build:
  stage: build
  script: 
   - docker build -t devintekhab-web-app


push:
  stage: push
  script:
  - docker login -u "$CI_REGISTERY_USER" -p "$CI_REGISTERY_PASSWORD" 
  - docker push devintekhab-web-app
  
deploy:
  stage: deploy
  script:
  - helm: upgrade -- install devintekhab-ngnix devintekhab/ngnix-ingress
