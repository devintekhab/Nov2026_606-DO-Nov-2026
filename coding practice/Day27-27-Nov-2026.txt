Day 27 - Serverless scaling with lambda and dynamodb

🔹 Ansible - Deploy Serverless lambda functions


 - name: deploy serverless lambda function 
   hosts: localhost
   tasks:
      - name: create lambda function
        command: >
         aws lambda create-function
         --function-name devintekhab-lambda 
         --runtime nodejs14.x
         --role arn:aws:iam::role/lambda-role
         --hanlder index.handler
         --zip-file fileb://lambda.zip

🔹 Kubernets - Deploy serverless using kubeless

apiVersion: kubless.io/v1beta1

   name: devintekhab-lambda
   namespace: default

  runtime: nodejs14
  handler: index.hanlder

🔹 Jenkinsfile - lambda function deployment Pipeline 

pipeline {
    agent any stages {

        stage("Build") {
            steps {
             echo "Buildig lambda function..."
             sh 'zip -r lambda.zip lambda/
            }
        }

        stage("deploy to lambda") {
            steps {
                script {
                    sh 'aws lambda update-function-code --function-name devintekhab-lambda --zip-file fileb://lambda.zip
                }
            }
        }
    }
}
