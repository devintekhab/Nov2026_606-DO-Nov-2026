tasks:
  - name: Configure kubectl for cluster 1 
   command: kubectl --context=us-east config use-context us-east-cluster
  
  - name: Deploy ngnix on cluster 1 
    command: kubectl apply -f kubernets/deployment.yaml

  - name: Configure kubectl fort cluster 2
    command: kubectl --context=us-west config use-context us-west-cluster

  - name: deploy ngnix on cluster 2
    command: kubectl apply -f kubernets/deployment.yaml

  name: devintekhab-deployment

  replicas : 3

  selector:
     matchlables: 
        app: devintekhab-app
    

 template:
   metadata:
     labels:
       app: devintekhab-app

spec:
 containers:
     -  name: ngnix
     -  image: ngnix
     -  ports:
          - containers: 8080
  

apiVersion: federation.k8s.io/v1beta1
kind: FederatedDeployment

metadata:
  name: devintekhab-federated-deployment
  namespace: default

spec:
  template:
    spec:
      replicas: 3

      selector:
        matchLabels:
          app: devintekhab-app

      template:
        metadata:
          labels:
            app: devintekhab-app

        spec:
          containers:
            - name: nginx
              image: nginx
              ports:
                - containerPort: 80
  

pipeline {
    agent any

    stages {
        stage('Deploy to Cluster 1') {
            steps {
                script {
                    sh '''
                        kubectl --context=us-east-cluster \
                        apply -f kubernetes/deployment.yaml
                    '''
                }
            }
        }

        stage('Deploy to Cluster 2') {
            steps {
                script {
                    sh '''
                        kubectl --context=us-west-cluster \
                        apply -f kubernetes/deployment.yaml
                    '''
                }
            }
        }
    }

    post {
        success {
            echo 'Nginx deployed successfully to both clusters!'
        }

        failure {
            echo 'Multi-cluster deployment failed!'
        }
    }
}
