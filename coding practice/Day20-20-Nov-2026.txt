🔹 Jenkinsfile - secure Credentials Managment 

pipeline {
    agent any Enviroment {
        REGISTERY_PASSWORD = credentials("docker-registery-password")
    }
    stages {
        stage("build"){
            steps {
                echo "building docker image...."
                docker.build("devintekhab-web-app")
            }
        }
        stage("push"){
            steps{
                script {
                    docker.withRegistry('https://docker.io',docker-credentials'){
                        docker.image('devintekhab-web-app').push('latest')
                    }
                }
            }
        }
    }
}
