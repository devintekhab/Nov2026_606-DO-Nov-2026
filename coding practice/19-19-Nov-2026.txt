🔹 Jenkinsfile - Inetgerating Promethous monitoring in Pipeline 

pipeline {
    agent any stages {
        stage("Build"){
            steps {
                echo "building docker image'
                docker.build('devintekhab-web-app')
            }
        }
        stage("test"){
            steps{
                echo "running test"
            }
        }
        stage("monitor deployment"){
            steps{
                script {
                    sh 'kubectl apply -f kubernets/deployment.yaml
                    sh 'kubectl/deployment.yaml'
                    sh 'kubectl get pods --watch'
                }
            }
        }
    }
}
